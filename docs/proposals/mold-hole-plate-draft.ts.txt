import { z } from "zod";
import type { PartModel } from "./step";
import { cross, normalize, sub, type Vec3 } from "./vec3";

/**
 * A mold is a rectangular plate carrying a grid of through holes, one per part.
 * Each hole sits at the centre of a `spacingX` x `spacingY` cell, so the plate
 * outline is exactly the cell grid and neighbouring cells share their seams.
 */
export const moldParamsSchema = z.object({
  /** Radial gap added around the part on each side. */
  clearance: z.number().min(0).max(10),
  /** Nominal part diameter the hole is cut for. */
  boreDiameter: z.number().min(0.1).max(500),
  spacingX: z.number().min(0.1).max(1000),
  spacingY: z.number().min(0.1).max(1000),
  columns: z.number().int().min(1).max(64),
  rows: z.number().int().min(1).max(64),
  thickness: z.number().min(0.5).max(200)
});

export type MoldParams = z.infer<typeof moldParamsSchema>;

export interface Mold {
  width: number;
  height: number;
  thickness: number;
  holeDiameter: number;
  /** Footprint of one grid cell; also the hole pitch. */
  cellWidth: number;
  cellHeight: number;
  /** Hole centres on the plate, measured from its lower-left corner. */
  holes: [number, number][];
}

const RING_SEGMENTS = 32;
/** Material left between a part and its cell wall. */
const WEB = 2;

const round = (value: number, decimals: number): number => Number(value.toFixed(decimals));
/** Snaps to sub-micron precision so coincident vertices of adjacent cells match exactly. */
const snap = (value: number): number => round(value, 6);

export function buildMold(params: MoldParams): Mold {
  const holes: [number, number][] = [];
  for (let row = 0; row < params.rows; row += 1) {
    for (let column = 0; column < params.columns; column += 1) {
      holes.push([(column + 0.5) * params.spacingX, (row + 0.5) * params.spacingY]);
    }
  }

  return {
    width: params.columns * params.spacingX,
    height: params.rows * params.spacingY,
    thickness: params.thickness,
    holeDiameter: round(params.boreDiameter + 2 * params.clearance, 3),
    cellWidth: params.spacingX,
    cellHeight: params.spacingY,
    holes
  };
}

/** Seeds the mold from the imported part: smallest bore, footprint-driven spacing. */
export function suggestParams(part: PartModel): MoldParams {
  const clearance = 0.2;
  const boreDiameter = round(part.boreDiameters[0] ?? 4, 2);
  const span = (axis: 0 | 1 | 2): number => Math.max(part.max[axis] - part.min[axis], boreDiameter);

  return {
    clearance,
    boreDiameter,
    spacingX: round(span(0) + 2 * clearance + WEB, 1),
    spacingY: round(span(1) + 2 * clearance + WEB, 1),
    columns: 4,
    rows: 3,
    thickness: round(Math.min(Math.max(span(2) / 2, 3), 40), 1)
  };
}

/** Parameters used before a part is imported. */
export const DEFAULT_PARAMS: MoldParams = suggestParams({
  edges: [],
  min: [0, 0, 0],
  max: [0, 0, 0],
  boreDiameters: []
});

/** Point on a cell rectangle at `angle`, so cell and hole rings share a parameter. */
function cellPoint(angle: number, halfWidth: number, halfHeight: number): [number, number] {
  const [cosine, sine] = [Math.cos(angle), Math.sin(angle)];
  const reach = Math.min(
    Math.abs(cosine) < 1e-9 ? Infinity : halfWidth / Math.abs(cosine),
    Math.abs(sine) < 1e-9 ? Infinity : halfHeight / Math.abs(sine)
  );
  return [cosine * reach, sine * reach];
}

/** Plate outline and hole circles as 3D polylines, for the viewport. */
export function moldWireframe(mold: Mold): Vec3[][] {
  const { width: w, height: h, thickness: t } = mold;
  const corners: [number, number][] = [[0, 0], [w, 0], [w, h], [0, h]];
  const outline = (z: number): Vec3[] => [...corners, corners[0]].map(([x, y]) => [x, y, z]);
  const ring = (x: number, y: number, z: number): Vec3[] =>
    Array.from({ length: RING_SEGMENTS + 1 }, (_, step) => {
      const angle = (2 * Math.PI * step) / RING_SEGMENTS;
      const radius = mold.holeDiameter / 2;
      return [x + radius * Math.cos(angle), y + radius * Math.sin(angle), z];
    });

  return [
    outline(0),
    outline(t),
    ...corners.map(([x, y]): Vec3[] => [[x, y, 0], [x, y, t]]),
    ...mold.holes.flatMap(([x, y]) => [ring(x, y, 0), ring(x, y, t)])
  ];
}

/** Plate outline and hole circles as a minimal R12 DXF, for 2D machining. */
export function toDxf(mold: Mold): string {
  const record = (...pairs: (string | number)[]): string => pairs.join("\n");
  const line = (x1: number, y1: number, x2: number, y2: number): string =>
    record(0, "LINE", 8, "PLATE", 10, x1, 20, y1, 11, x2, 21, y2);

  return record(
    0, "SECTION", 2, "ENTITIES",
    line(0, 0, mold.width, 0),
    line(mold.width, 0, mold.width, mold.height),
    line(mold.width, mold.height, 0, mold.height),
    line(0, mold.height, 0, 0),
    ...mold.holes.map(([x, y]) => record(0, "CIRCLE", 8, "HOLES", 10, x, 20, y, 40, mold.holeDiameter / 2)),
    0, "ENDSEC", 0, "EOF", ""
  );
}

/** The plate as a closed ASCII STL solid. */
export function toStl(mold: Mold, name: string): string {
  const facets: string[] = [];
  const facet = (a: Vec3, b: Vec3, c: Vec3): void => {
    const [nx, ny, nz] = normalize(cross(sub(b, a), sub(c, a)));
    facets.push(
      [
        `facet normal ${nx} ${ny} ${nz}`,
        "outer loop",
        ...[a, b, c].map((point) => `vertex ${point.map(snap).join(" ")}`),
        "endloop",
        "endfacet"
      ].join("\n")
    );
  };
  const quad = (a: Vec3, b: Vec3, c: Vec3, d: Vec3): void => {
    facet(a, b, c);
    facet(a, c, d);
  };

  const top = mold.thickness;
  const radius = mold.holeDiameter / 2;
  const halfWidth = mold.cellWidth / 2;
  const halfHeight = mold.cellHeight / 2;

  // Ring angles plus the cell corners, so every segment lies on a single cell
  // edge. Corner angles keep the outer wall closed; the symmetric ring keeps
  // neighbouring cells sharing their seam vertices exactly.
  const corners = [[1, 1], [-1, 1], [-1, -1], [1, -1]].map(
    ([signX, signY]) => (Math.atan2(signY * halfHeight, signX * halfWidth) + 2 * Math.PI) % (2 * Math.PI)
  );
  const angles = [
    ...new Set([...Array.from({ length: RING_SEGMENTS }, (_, step) => (2 * Math.PI * step) / RING_SEGMENTS), ...corners])
  ].sort((a, b) => a - b);

  const near = (value: number, target: number): boolean => Math.abs(value - target) < 1e-6;
  const onPlateEdge = (a: [number, number], b: [number, number]): boolean =>
    (near(a[0], 0) && near(b[0], 0)) ||
    (near(a[0], mold.width) && near(b[0], mold.width)) ||
    (near(a[1], 0) && near(b[1], 0)) ||
    (near(a[1], mold.height) && near(b[1], mold.height));

  const at = (point: [number, number], z: number): Vec3 => [point[0], point[1], z];

  for (const [centreX, centreY] of mold.holes) {
    angles.forEach((angle, index) => {
      const ends = [angle, angles[(index + 1) % angles.length]];
      const [holeA, holeB] = ends.map(
        (value): [number, number] => [centreX + radius * Math.cos(value), centreY + radius * Math.sin(value)]
      );
      const [cellA, cellB] = ends.map((value): [number, number] => {
        const [offsetX, offsetY] = cellPoint(value, halfWidth, halfHeight);
        return [centreX + offsetX, centreY + offsetY];
      });

      quad(at(cellA, top), at(cellB, top), at(holeB, top), at(holeA, top));
      quad(at(holeA, 0), at(holeB, 0), at(cellB, 0), at(cellA, 0));
      quad(at(holeA, 0), at(holeA, top), at(holeB, top), at(holeB, 0));
      if (onPlateEdge(cellA, cellB)) quad(at(cellA, 0), at(cellB, 0), at(cellB, top), at(cellA, top));
    });
  }

  return [`solid ${name}`, ...facets, `endsolid ${name}`, ""].join("\n");
}
